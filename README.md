# Lambda-crud
Codigos Utilizados : 
Leer "producto_002"
{
  "action": "read",
  "key": {
    "id": "producto_002"
  }
}

Actualizar "producto_002"
{
  "action": "update",
  "key": {
    "id": "producto_002"
  },
  "item": {
    "Nombre": "Pelota_huellitas",
    "Precio": 9.00,
    "Stock": 50
  }
}

Eliminar "producto_002"
{
  "action": "delete",
  "key": {
    "id": "producto_002"
  }
}

{
  "action": "create",
  "item": {
    "id": "producto_002",

  }
